//@version=5
// ============================================================================
// Core Invest Viewer  |  Capitalife Terminal  |  v2.0
// ============================================================================
// Presets (Dropdown) und deren Chart-Symbol:
//   QQQ_PASSIVE  45%  -> QQQ  (Buy & Hold)
//   GLD          25%  -> GLD  (Buy & Hold)
//   SPMO          5%  -> SPMO (Buy & Hold)
//   SPY           5%  -> SPY  (Buy & Hold)
//   QQQ_PINE_1   10%  -> QQQ  (White Swan NAS EMA: SMA400/SMA5, SL=25%, TP=2%)
//   COPPER_HG     5%  -> HG1! (EMA + Valuation PRO: EMA20/50,  SL=2%,  TP=4%)
//   CHF_6S        5%  -> 6S1! (EMA + Valuation PRO: EMA20/50,  SL=2%,  TP=4%)
//
// Quellen:  QQQ_pine1.txt  |  pine2.txt
// Execution: process_orders_on_close=false  ->  NextOpen (TradingView Default)
// ============================================================================

strategy("Core Invest Viewer",
     overlay                 = true,
     initial_capital         = 10000,
     default_qty_type        = strategy.percent_of_equity,
     default_qty_value       = 100,
     process_orders_on_close = false,
     calc_on_order_fills     = false,
     pyramiding              = 0,
     commission_type         = strategy.commission.cash_per_contract,
     commission_value        = 0.005)

// ============================================================================
// 1) PRESET
// ============================================================================
// Wichtig: input.string options[] braucht Literal-Strings (keine Variablen).

G_PRE = "1) Preset"

preset = input.string(
     defval  = "QQQ_PINE_1",
     title   = "Komponente",
     options = ["QQQ_PASSIVE", "GLD", "SPMO", "SPY",
                "QQQ_PINE_1", "COPPER_HG", "CHF_6S"],
     tooltip = "Chart-Symbol passend zum Preset laden:\n"  +
               "QQQ_PASSIVE -> QQQ  |  GLD -> GLD\n"      +
               "SPMO -> SPMO        |  SPY -> SPY\n"       +
               "QQQ_PINE_1 -> QQQ   |  COPPER_HG -> HG1!\n" +
               "CHF_6S -> 6S1!",
     group   = G_PRE)

isPassive = preset == "QQQ_PASSIVE" or preset == "GLD" or
            preset == "SPMO"        or preset == "SPY"
isPine1   = preset == "QQQ_PINE_1"
isEMA     = preset == "COPPER_HG"  or preset == "CHF_6S"

// ============================================================================
// 2) ZEITFILTER
// ============================================================================
G_TIME = "2) Zeit"

startTime = input.time(timestamp("01 Jan 1995 00:00 +0000"), "Start", group=G_TIME)
endTime   = input.time(timestamp("01 Jan 2099 00:00 +0000"), "End",   group=G_TIME)
inDate    = time >= startTime and time <= endTime

// ============================================================================
// 3) QQQ_PINE_1 — White Swan NAS EMA
//    Quelle: QQQ_pine1.txt
//    Entry : close > SMA(400) AND close < SMA(5)
//    Exit  : close > SMA(5) | SL 25% | TP 2%
//    Exec  : NextOpen  (process_orders_on_close=false)
// ============================================================================
G_P1 = "3) QQQ_PINE_1  [QQQ_pine1.txt]"

p1_ma1 = input.int(400, "SMA Trend (MA1)", minval=1, group=G_P1,
     tooltip="Langfristiger Bull-Trend-Filter. Default: 400")
p1_ma2 = input.int(5,   "SMA Dip   (MA2)", minval=1, group=G_P1,
     tooltip="Kurzfristiger Dip-Filter. Default: 5")
p1_sl  = input.float(25.0, "Stop Loss %",   minval=0.1, step=1.0, group=G_P1,
     tooltip="Safety Stop. Default: 25%")
p1_tp  = input.float(2.0,  "Take Profit %", minval=0.1, step=0.5, group=G_P1,
     tooltip="Safety Profit. Default: 2%")

// ============================================================================
// 4) EMA SLEEVE — COPPER_HG + CHF_6S
//    Quelle: pine2.txt  —  EMA + Valuation Strategy PRO MTF + Regime
//    Entry : EMA(20) > EMA(50) AND close > EMA(20)  +  Valuation-Window
//    Exit  : SL 2% | TP 4% | opposite Valuation | EMA-Flip (optional)
//    Exec  : NextOpen  (process_orders_on_close=false)
// ============================================================================
G_EMA = "4) EMA Sleeve  [pine2.txt]"

e_fast       = input.int(20,  "EMA Fast",        minval=1,   group=G_EMA)
e_slow       = input.int(50,  "EMA Slow",        minval=1,   group=G_EMA)
e_sl         = input.float(2.0, "Stop Loss %",   minval=0.1, step=0.1, group=G_EMA)
e_tp         = input.float(4.0, "Take Profit %", minval=0.1, step=0.1, group=G_EMA)
exitOppVal   = input.bool(true,  "Exit: opposite Valuation", group=G_EMA,
     tooltip="exitOnOppositeVal=true in pine2.txt")
exitEMAFlip  = input.bool(false, "Exit: EMA Flip",           group=G_EMA,
     tooltip="exitOnEMAFlip=false in pine2.txt")
onePerDay    = input.bool(true,  "Max 1 Entry / Tag",         group=G_EMA,
     tooltip="oneEntryPerDay=true in pine2.txt")
minHold      = input.int(0,  "Min Hold Bars", minval=0,       group=G_EMA,
     tooltip="minHoldBars=0 in pine2.txt")

// ============================================================================
// 5) VALUATION FILTER  (EMA Sleeves)
//    Quelle: pine2.txt  Sections VALUATION + COMPARISON SYMBOLS
// ============================================================================
G_VAL = "5) Valuation  [pine2.txt]"

useVal    = input.bool(true,   "Valuation aktiv",      group=G_VAL)
valFast   = input.int(10,  "Val Fast",                 group=G_VAL,
     tooltip="valFast=10 in pine2.txt")
valSlow   = input.int(20,  "Val Slow",                 group=G_VAL,
     tooltip="valSlow=20 in pine2.txt")
rescaleLB = input.int(100, "Rescale Lookback",         group=G_VAL,
     tooltip="rescaleLookback=100 in pine2.txt")
valUp     = input.float( 75.0, "Upper Threshold",      group=G_VAL,
     tooltip="valUpper=75 in pine2.txt")
valDn     = input.float(-75.0, "Lower Threshold",      group=G_VAL,
     tooltip="valLower=-75 in pine2.txt")
valReq    = input.string("1 von 3", "Symbol Requirement",
     options=["1 von 3", "2 von 3", "3 von 3"],        group=G_VAL,
     tooltip="mode='1 von 3' in pine2.txt")
entryWin  = input.int(3, "Entry Window Bars", minval=0, group=G_VAL,
     tooltip="entryBars=3 in pine2.txt")

// Vergleichssymbole — exakt wie pine2.txt
sym1 = input.symbol("TVC:DXY",    "Symbol 1 (DXY)",   group=G_VAL)
sym2 = input.symbol("COMEX:GC1!", "Symbol 2 (Gold)",   group=G_VAL)
sym3 = input.symbol("CBOT:ZB1!",  "Symbol 3 (T-Bond)", group=G_VAL)

// ============================================================================
// HILFSFUNKTIONEN
// ============================================================================
f_perf(float src, int len) =>
    float base = nz(src[len])
    base != 0.0 ? (src - base) / base * 100.0 : 0.0

f_rescale(float v, int l) =>
    float hi  = ta.highest(v, l)
    float lo  = ta.lowest(v, l)
    float rng = math.max(hi - lo, 0.0001)
    (v - lo) / rng * 200.0 - 100.0

// ============================================================================
// INDIKATOREN — PINE 1
// ============================================================================
ma1 = ta.sma(close, p1_ma1)
ma2 = ta.sma(close, p1_ma2)

// ============================================================================
// INDIKATOREN — EMA SLEEVE
// ============================================================================
emaFast = ta.ema(close, e_fast)
emaSlow = ta.ema(close, e_slow)

emaLongOk = emaFast > emaSlow and close > emaFast

// ============================================================================
// VALUATION — externe Daten
// ============================================================================
s1 = request.security(sym1, timeframe.period, close, lookahead=barmerge.lookahead_off)
s2 = request.security(sym2, timeframe.period, close, lookahead=barmerge.lookahead_off)
s3 = request.security(sym3, timeframe.period, close, lookahead=barmerge.lookahead_off)

bFast = f_perf(close, valFast)
bSlow = f_perf(close, valSlow)

r1f = f_rescale(bFast - f_perf(s1, valFast), rescaleLB)
r2f = f_rescale(bFast - f_perf(s2, valFast), rescaleLB)
r3f = f_rescale(bFast - f_perf(s3, valFast), rescaleLB)
r1s = f_rescale(bSlow - f_perf(s1, valSlow), rescaleLB)
r2s = f_rescale(bSlow - f_perf(s2, valSlow), rescaleLB)
r3s = f_rescale(bSlow - f_perf(s3, valSlow), rescaleLB)

req = valReq == "1 von 3" ? 1 : valReq == "2 von 3" ? 2 : 3

dnFast = (r1f < valDn ? 1 : 0) + (r2f < valDn ? 1 : 0) + (r3f < valDn ? 1 : 0) >= req
upFast = (r1f > valUp ? 1 : 0) + (r2f > valUp ? 1 : 0) + (r3f > valUp ? 1 : 0) >= req
dnSlow = (r1s < valDn ? 1 : 0) + (r2s < valDn ? 1 : 0) + (r3s < valDn ? 1 : 0) >= req
upSlow = (r1s > valUp ? 1 : 0) + (r2s > valUp ? 1 : 0) + (r3s > valUp ? 1 : 0) >= req

// valSignalMode = "All": fast OR slow reicht
valLong  = dnFast or dnSlow   // Asset billig vs. Vergleich -> Long-Signal
valShort = upFast or upSlow   // Asset teuer  vs. Vergleich -> Short-/Exit-Signal

// Entry-Window tracken (entryBars=3)
var int lastLongBar  = na
var int lastShortBar = na

if valLong
    lastLongBar  := bar_index
if valShort
    lastShortBar := bar_index

inLongWin  = not na(lastLongBar)  and (bar_index - lastLongBar)  <= entryWin
inShortWin = not na(lastShortBar) and (bar_index - lastShortBar) <= entryWin

valEntryOk = not useVal or inLongWin
valExitOk  = useVal and exitOppVal and inShortWin

// ============================================================================
// DAILY ENTRY LOCK
// ============================================================================
var bool tradedToday = false
if ta.change(time("D")) != 0
    tradedToday := false
dailyOk = not onePerDay or not tradedToday

// ============================================================================
// TRADE STATE
// ============================================================================
inTrade = strategy.position_size > 0

var int entryBar = na
if inTrade and strategy.position_size[1] == 0
    entryBar := bar_index

barsHeld = inTrade and not na(entryBar) ? bar_index - entryBar : 0
holdOk   = barsHeld >= minHold

// ============================================================================
// PASSIVE — BUY & HOLD
// ============================================================================
if isPassive and inDate and barstate.isconfirmed and not inTrade
    strategy.entry("BuyHold", strategy.long)

// ============================================================================
// QQQ_PINE_1 — White Swan NAS EMA
// ============================================================================
p1Buy  = isPine1 and inDate and barstate.isconfirmed and not inTrade and
         not na(ma1) and not na(ma2) and close > ma1 and close < ma2

p1Sell = isPine1 and inDate and barstate.isconfirmed and inTrade and
         not na(ma2) and close > ma2

if p1Buy
    strategy.entry("Pine1", strategy.long)

if p1Sell
    strategy.close("Pine1", comment="Signal Exit")

if isPine1 and inTrade
    strategy.exit("P1 SL/TP", from_entry="Pine1",
         stop    = strategy.position_avg_price * (1.0 - p1_sl / 100.0),
         limit   = strategy.position_avg_price * (1.0 + p1_tp / 100.0))

// ============================================================================
// EMA SLEEVE — COPPER_HG + CHF_6S
// ============================================================================
emaBuy    = isEMA and inDate and barstate.isconfirmed and not inTrade and
            emaLongOk and valEntryOk and dailyOk

emaValX   = isEMA and inTrade and holdOk and valExitOk  and barstate.isconfirmed
emaFlipX  = isEMA and inTrade and holdOk and exitEMAFlip and
            emaFast < emaSlow and barstate.isconfirmed

if emaBuy
    strategy.entry("EMA", strategy.long)
    tradedToday := true

if emaValX
    strategy.close("EMA", comment="Opposite Val")

if emaFlipX
    strategy.close("EMA", comment="EMA Flip")

if isEMA and inTrade
    strategy.exit("EMA SL/TP", from_entry="EMA",
         stop  = strategy.position_avg_price * (1.0 - e_sl / 100.0),
         limit = strategy.position_avg_price * (1.0 + e_tp / 100.0))

// ============================================================================
// PLOTS — Indikatorlinien
// ============================================================================
plot(isPine1 ? ma1 : na, "SMA 400", color=color.new(color.blue,   0), linewidth=2)
plot(isPine1 ? ma2 : na, "SMA 5",   color=color.new(color.orange, 0), linewidth=1)
plot(isEMA ? emaFast : na, "EMA 20", color=color.new(color.orange, 0), linewidth=2)
plot(isEMA ? emaSlow : na, "EMA 50", color=color.new(color.blue,   0), linewidth=2)

// ============================================================================
// PLOTS — Entry / Exit Shapes
// ============================================================================
plotshape(p1Buy,
     title="Pine1 Entry", style=shape.triangleup, location=location.belowbar,
     color=color.lime, size=size.small, text="E")

plotshape(p1Sell,
     title="Pine1 Exit", style=shape.triangledown, location=location.abovebar,
     color=color.red, size=size.small, text="X")

plotshape(emaBuy,
     title="EMA Entry", style=shape.triangleup, location=location.belowbar,
     color=color.lime, size=size.small, text="E")

plotshape(emaValX or emaFlipX,
     title="EMA Exit", style=shape.triangledown, location=location.abovebar,
     color=color.orange, size=size.small, text="X")

// ============================================================================
// PLOTS — Hintergrundfarbe (Position offen)
// ============================================================================
bgcolor(inTrade and not isPassive ? color.new(color.green, 90) : na,
     title="In Position")

// Valuation-Long-Signal-Fenster sichtbar machen (nur bei EMA Presets)
bgcolor(isEMA and inLongWin  ? color.new(color.blue,  95) : na, title="Val Long Window")
bgcolor(isEMA and inShortWin ? color.new(color.red,   95) : na, title="Val Short Window")

// ============================================================================
// INFO-TABELLE (top right)
// ============================================================================
f_weight(string p) =>
    p == "QQQ_PASSIVE" ? "45%" :
     p == "GLD"         ? "25%" :
     p == "QQQ_PINE_1"  ? "10%" : "5%"

f_logic(string p) =>
    p == "QQQ_PINE_1"  ? "SMA(400)/SMA(5)" :
     p == "COPPER_HG" or p == "CHF_6S" ? "EMA(20)/EMA(50)" : "Buy & Hold"

f_sltp(string p) =>
    p == "QQQ_PINE_1"  ? "SL 25% / TP 2%" :
     p == "COPPER_HG" or p == "CHF_6S" ? "SL 2% / TP 4%" : "--"

var table tbl = table.new(position.top_right, 2, 7,
     border_width=1, border_color=color.new(color.gray, 60))

if barstate.islast
    C0  = color.rgb(15, 15, 30)
    C1  = color.rgb(25, 25, 45)
    FG  = color.white
    GR  = color.new(color.lime,   0)
    YE  = color.new(color.yellow, 0)
    RD  = color.new(color.red,    0)

    winRate  = strategy.closedtrades > 0 ?
         str.tostring(math.round(strategy.wintrades / strategy.closedtrades * 100, 1)) + "%" : "--"
    netPct   = str.tostring(math.round((strategy.equity / strategy.initial_capital - 1) * 100, 2)) + "%"
    netColor = strategy.equity >= strategy.initial_capital ? GR : RD

    table.cell(tbl, 0, 0, "Core Invest v2.0",        text_color=GR,  bgcolor=C0, text_halign=text.align_left,  text_size=size.small)
    table.cell(tbl, 1, 0, preset,                     text_color=GR,  bgcolor=C0, text_halign=text.align_right, text_size=size.small)
    table.cell(tbl, 0, 1, "Gewicht",                  text_color=FG,  bgcolor=C1, text_halign=text.align_left,  text_size=size.tiny)
    table.cell(tbl, 1, 1, f_weight(preset),           text_color=YE,  bgcolor=C1, text_halign=text.align_right, text_size=size.tiny)
    table.cell(tbl, 0, 2, "Logik",                    text_color=FG,  bgcolor=C1, text_halign=text.align_left,  text_size=size.tiny)
    table.cell(tbl, 1, 2, f_logic(preset),            text_color=FG,  bgcolor=C1, text_halign=text.align_right, text_size=size.tiny)
    table.cell(tbl, 0, 3, "SL / TP",                  text_color=FG,  bgcolor=C1, text_halign=text.align_left,  text_size=size.tiny)
    table.cell(tbl, 1, 3, f_sltp(preset),             text_color=FG,  bgcolor=C1, text_halign=text.align_right, text_size=size.tiny)
    table.cell(tbl, 0, 4, "Closed Trades",            text_color=FG,  bgcolor=C1, text_halign=text.align_left,  text_size=size.tiny)
    table.cell(tbl, 1, 4, str.tostring(strategy.closedtrades), text_color=GR, bgcolor=C1, text_halign=text.align_right, text_size=size.tiny)
    table.cell(tbl, 0, 5, "Win Rate",                 text_color=FG,  bgcolor=C1, text_halign=text.align_left,  text_size=size.tiny)
    table.cell(tbl, 1, 5, winRate,                    text_color=GR,  bgcolor=C1, text_halign=text.align_right, text_size=size.tiny)
    table.cell(tbl, 0, 6, "Net P&L",                  text_color=FG,  bgcolor=C1, text_halign=text.align_left,  text_size=size.tiny)
    table.cell(tbl, 1, 6, netPct,                     text_color=netColor, bgcolor=C1, text_halign=text.align_right, text_size=size.tiny)
